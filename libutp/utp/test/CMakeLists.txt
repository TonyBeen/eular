function(utp_apply_toolchain_flags target flags_variable)
    set(flags ${${flags_variable}})
    if(NOT flags)
        return()
    endif()
    target_compile_options(${target} PRIVATE ${flags})
    if(CMAKE_VERSION VERSION_LESS "3.13")
        string(REPLACE ";" " " link_flags "${flags}")
        set_property(TARGET ${target} APPEND_STRING PROPERTY LINK_FLAGS " ${link_flags}")
    else()
        target_link_options(${target} PRIVATE ${flags})
    endif()
endfunction()

function(utp_append_link_flags target)
    if(CMAKE_VERSION VERSION_LESS "3.13")
        string(REPLACE ";" " " link_flags "${ARGN}")
        set_property(TARGET ${target} APPEND_STRING PROPERTY LINK_FLAGS " ${link_flags}")
    else()
        target_link_options(${target} PRIVATE ${ARGN})
    endif()
endfunction()

function(utp_configure_c_test target)
    target_link_libraries(${target} PRIVATE utp event_core)
    target_include_directories(${target} PRIVATE "${UTP_ROOT_DIR}/src")
    utp_apply_toolchain_flags(${target} UTP_SANITIZER_FLAGS)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow
            -Wstrict-prototypes -Wmissing-prototypes -Wformat=2 -UNDEBUG)
    elseif(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /WX /UNDEBUG)
    endif()
    add_test(NAME ${target} COMMAND ${target})
endfunction()

function(utp_configure_cpp_test target)
    target_link_libraries(${target} PRIVATE utp event_core)
    target_include_directories(${target} PRIVATE "${UTP_ROOT_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}"
        "${UTP_ROOT_DIR}/../3rd")
    target_compile_definitions(${target} PRIVATE ${UTP_SOCKET_BATCH_DEFINITIONS})
    utp_apply_toolchain_flags(${target} UTP_SANITIZER_FLAGS)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow -Wformat=2 -UNDEBUG)
    elseif(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /WX /EHsc /UNDEBUG)
    endif()
    add_test(NAME ${target} COMMAND ${target})
endfunction()
