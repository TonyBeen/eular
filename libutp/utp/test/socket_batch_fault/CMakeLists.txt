if(POLICY CMP0079)
    cmake_policy(SET CMP0079 NEW)

add_library(utp_socket_batch_hook SHARED socket_batch_hook.c)
target_include_directories(utp_socket_batch_hook PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(utp_socket_batch_hook PRIVATE dl)
target_compile_definitions(utp_socket_batch_hook PRIVATE ${UTP_SOCKET_BATCH_DEFINITIONS})

target_include_directories(utp_transport_integration_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(utp_transport_integration_test PRIVATE utp_socket_batch_hook)
if(UTP_HAVE_SENDMMSG)
    utp_append_link_flags(utp_transport_integration_test "-Wl,--wrap=sendmmsg")
endif()
if(UTP_HAVE_RECVMMSG)
    utp_append_link_flags(utp_transport_integration_test "-Wl,--wrap=recvmmsg")
endif()
